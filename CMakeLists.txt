cmake_minimum_required(VERSION 3.28)
set(versionFile "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${versionFile}")
file(READ "${versionFile}" LYRA_VERSION)
string(STRIP "${LYRA_VERSION}" LYRA_VERSION)
if(NOT LYRA_VERSION MATCHES "^([0-9]+\\.[0-9]+\\.[0-9]+)(-[0-9A-Za-z-]+(\\.[0-9A-Za-z-]+)*)?$")
    message(FATAL_ERROR "VERSION must contain x.y.z or x.y.z-prerelease.")
endif()
project(Lyra VERSION "${CMAKE_MATCH_1}" LANGUAGES C CXX)

if(CMAKE_CROSSCOMPILING OR NOT CMAKE_SYSTEM_NAME MATCHES "^(Linux|Windows)$" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    message(FATAL_ERROR "Lyra builds natively on x86-64 Linux and Windows.")
endif()

if(NOT CMAKE_GENERATOR STREQUAL "Ninja" OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR
   NOT CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "GNU" OR CMAKE_CXX_COMPILER_VERSION VERSION_LESS 18)
    message(FATAL_ERROR "Use a Lyra preset with Ninja and the Clang GNU driver (clang++).")
endif()
if(WIN32)
    if(NOT CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC")
        message(FATAL_ERROR "Build the Windows compiler with the MSVC LLVM SDK and the matching host Clang.")
    endif()
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    add_link_options(-fuse-ld=lld)
elseif(NOT CMAKE_CXX_COMPILER_VERSION VERSION_LESS 19)
    message(FATAL_ERROR "The Linux presets use Clang 18.")
endif()
execute_process(COMMAND "${CMAKE_MAKE_PROGRAM}" --version OUTPUT_VARIABLE ninjaVersion OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(ninjaVersion VERSION_LESS 1.11)
    message(FATAL_ERROR "C++ module dependency scanning requires Ninja 1.11 or newer.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(LLVM 18.1 REQUIRED CONFIG)
include(CTest)
include(cmake/LyraSdk.cmake)
if(WIN32)
    include(cmake/LyraWindows.cmake)
    lyra_stage_windows_sdk()
    lyra_relocate_llvm_dia()
else()
    include(cmake/LyraRuntime.cmake)
    lyra_add_runtime()
    lyra_stage_linux_sdk()
endif()

add_executable(lyra compiler/main.cpp compiler/lsp.cpp compiler/source.cpp compiler/parser.cpp compiler/check.cpp compiler/lowering.cpp compiler/build.cpp)
include(cmake/LyraCImport.cmake)
lyra_stage_c_frontend(lyra)
set_source_files_properties(compiler/main.cpp PROPERTIES COMPILE_DEFINITIONS "LYRA_TOOLCHAIN_VERSION=\"${LYRA_VERSION}\"")
target_sources(lyra PRIVATE FILE_SET CXX_MODULES FILES
    compiler/platform.cppm compiler/source.cppm compiler/syntax.cppm compiler/check.cppm compiler/lowering.cppm compiler/build.cppm
    compiler/packages.cppm compiler/modules.cppm compiler/generics.cppm compiler/async.cppm compiler/foreign.cppm compiler/sums.cppm compiler/macros.cppm compiler/lsp.cppm)
target_sources(lyra PRIVATE FILE_SET CXX_MODULES FILES compiler/traits.cppm compiler/patterns.cppm compiler/numbers.cppm compiler/initialization.cppm)
target_sources(lyra PRIVATE FILE_SET CXX_MODULES FILES compiler/collections.cppm compiler/conversions.cppm compiler/exceptions.cppm compiler/closures.cppm)
target_sources(lyra PRIVATE FILE_SET CXX_MODULES FILES compiler/ctfe.cppm compiler/macro_syntax.cppm compiler/format.cppm)
target_sources(lyra PRIVATE FILE_SET CXX_MODULES FILES compiler/c_import.cppm compiler/native_packages.cppm)
target_sources(lyra PRIVATE FILE_SET CXX_MODULES FILES compiler/build_cache.cppm)
target_sources(lyra PRIVATE FILE_SET CXX_MODULES FILES compiler/doc.cppm)
lyra_stage_standard_library()
add_dependencies(lyra lyraRuntime)
if(NOT WIN32)
    add_dependencies(lyra lyraHttpClient lyraHttpTls lyraCompression lyraRegistryDownload)
endif()
target_compile_options(lyra PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(lyra PRIVATE include)
target_include_directories(lyra SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
if(WIN32)
    set_property(TARGET lyra PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
    target_link_libraries(lyra PRIVATE LLVMCore LLVMPasses LLVMX86CodeGen LLVMObject)
else()
    target_link_libraries(lyra PRIVATE LLVM)
    target_link_options(lyra PRIVATE -Wl,--disable-new-dtags)
    set_target_properties(lyra PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/../lib/lyra/host")
endif()
set_target_properties(lyra PROPERTIES CXX_EXTENSIONS OFF RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(WIN32)
    install(TARGETS lyra RUNTIME DESTINATION bin)
else()
    set_target_properties(lyra PROPERTIES OUTPUT_NAME lyra.real)
    configure_file(cmake/lyra-launcher.sh.in "${CMAKE_BINARY_DIR}/bin/lyra" @ONLY)
    file(CHMOD "${CMAKE_BINARY_DIR}/bin/lyra" PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
    install(TARGETS lyra RUNTIME DESTINATION bin)
    install(PROGRAMS "${CMAKE_BINARY_DIR}/bin/lyra" DESTINATION bin)
endif()
install(FILES include/lyra/string.h DESTINATION include/lyra)
install(FILES cmake/third-party.md DESTINATION share/lyra/licenses)

if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    if(WIN32)
        find_program(LYRA_DEBUG_INFO_TOOL NAMES llvm-pdbutil HINTS ${LLVM_TOOLS_BINARY_DIR} REQUIRED)
    else()
        find_program(LYRA_DEBUG_INFO_TOOL NAMES llvm-dwarfdump HINTS ${LLVM_TOOLS_BINARY_DIR} REQUIRED)
    endif()
    add_test(NAME frontend COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/compiler_cli.py $<TARGET_FILE:lyra> Frontend)
    add_test(NAME native COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/compiler_cli.py $<TARGET_FILE:lyra> Native)
    add_test(NAME debug-info COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/debug_info_cli.py $<TARGET_FILE:lyra> ${LYRA_DEBUG_INFO_TOOL})
    add_test(NAME objects COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/objects_cli.py $<TARGET_FILE:lyra>)
    add_test(NAME closures COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/closures_cli.py $<TARGET_FILE:lyra>)
    add_test(NAME ffi COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/ffi_cli.py $<TARGET_FILE:lyra> ${CMAKE_COMMAND})
    add_test(NAME http COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/http_cli.py $<TARGET_FILE:lyra>)
    foreach(suite IN ITEMS packages modules doc generics async generics_tasks io io_fs sqlite sums macros jsonlines conversions overloads defer exceptions text text_collections_std syntax inference const json_codec json_builder toml reflection lsp stdlib_daily application_std http_client http_tls http2 http_multipart http_compression compression os_time format)
        add_test(NAME ${suite} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/${suite}_cli.py $<TARGET_FILE:lyra>)
    endforeach()
    add_test(NAME collections COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/collections_cli.py $<TARGET_FILE:lyra>)
    add_test(NAME context COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/context_cli.py $<TARGET_FILE:lyra>)
    add_test(NAME installed COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/installed_cli.py --build-tree ${CMAKE_COMMAND} ${CMAKE_BINARY_DIR})
    add_test(NAME sugar COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/sugar_cli.py $<TARGET_FILE:lyra>)
    get_property(cliSuites DIRECTORY PROPERTY TESTS)
    set_tests_properties(${cliSuites} PROPERTIES TIMEOUT 180)
    set_tests_properties(http PROPERTIES TIMEOUT 360)
    set_tests_properties(macros PROPERTIES TIMEOUT 600)
    set_tests_properties(installed PROPERTIES TIMEOUT 300)
    include(cmake/LyraRuntimeTests.cmake)
    lyra_register_runtime_tests("${CMAKE_BINARY_DIR}/runtime-tests")
endif()
