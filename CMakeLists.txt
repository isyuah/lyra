cmake_minimum_required(VERSION 3.28)
project(Lyra VERSION 0.1.0 LANGUAGES C CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    message(FATAL_ERROR "This compiler slice targets x86-64 Linux. On Windows configure it inside WSL.")
endif()

if(NOT CMAKE_GENERATOR STREQUAL "Ninja" OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR
   CMAKE_CXX_COMPILER_VERSION VERSION_LESS 18 OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_LESS 19)
    message(FATAL_ERROR "Use cmake --preset dev or release (Ninja and Clang 18 required).")
endif()
execute_process(COMMAND "${CMAKE_MAKE_PROGRAM}" --version OUTPUT_VARIABLE ninjaVersion OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(ninjaVersion VERSION_LESS 1.11)
    message(FATAL_ERROR "C++ module dependency scanning requires Ninja 1.11 or newer.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(LLVM 18.1 REQUIRED CONFIG)
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
find_library(gcArchive NAMES libgc.a REQUIRED)
find_library(uvArchive NAMES libuv.a libuv_a.a REQUIRED)
find_library(jsonArchive NAMES libjansson.a REQUIRED)
configure_file("${gcArchive}" lib/libgc.a COPYONLY)
configure_file("${uvArchive}" lib/libuv.a COPYONLY)
configure_file("${jsonArchive}" lib/libjansson.a COPYONLY)
add_library(lyraRuntime STATIC runtime/core.cpp runtime/strings.cpp runtime/managed.cpp runtime/tasks.cpp
    runtime/io.cpp runtime/io_args.cpp runtime/sqlite.cpp runtime/sqlite_api.cpp runtime/json.cpp)
include(cmake/LyraHttp.cmake)
target_sources(lyraRuntime PRIVATE runtime/lists.cpp runtime/exceptions.cpp runtime/context.cpp)
lyra_enable_http(lyraRuntime)
target_compile_options(lyraRuntime PRIVATE -fno-exceptions -fno-rtti)
target_include_directories(lyraRuntime PUBLIC runtime PRIVATE include)
target_link_libraries(lyraRuntime PUBLIC SQLite::SQLite3 INTERFACE "${gcArchive}" "${uvArchive}" "${jsonArchive}" Threads::Threads ${CMAKE_DL_LIBS} rt)
set_target_properties(lyraRuntime PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")

add_executable(lyra compiler/main.cpp compiler/source.cpp compiler/parser.cpp compiler/check.cpp compiler/lowering.cpp compiler/build.cpp)
set_source_files_properties(compiler/main.cpp PROPERTIES COMPILE_DEFINITIONS "LYRA_TOOLCHAIN_VERSION=\"${PROJECT_VERSION}-alpha.1\"")
target_sources(lyra PRIVATE FILE_SET CXX_MODULES FILES
    compiler/source.cppm compiler/syntax.cppm compiler/check.cppm compiler/lowering.cppm compiler/build.cppm
    compiler/packages.cppm compiler/modules.cppm compiler/generics.cppm compiler/async.cppm compiler/foreign.cppm compiler/sums.cppm compiler/macros.cppm)
file(GLOB_RECURSE standardSources CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/std" "std/*.ly")
target_sources(lyra PRIVATE FILE_SET CXX_MODULES FILES compiler/collections.cppm compiler/conversions.cppm compiler/exceptions.cppm compiler/closures.cppm)
foreach(source IN LISTS standardSources)
    configure_file("std/${source}" "share/lyra/std/${source}" COPYONLY)
endforeach()
install(DIRECTORY std/ DESTINATION share/lyra/std FILES_MATCHING PATTERN "*.ly")
add_dependencies(lyra lyraRuntime)
target_compile_options(lyra PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(lyra SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
target_link_libraries(lyra PRIVATE LLVM)
set_target_properties(lyra PROPERTIES CXX_EXTENSIONS OFF RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
install(TARGETS lyra lyraRuntime RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
install(FILES include/lyra/string.h DESTINATION include/lyra)
install(FILES "${CMAKE_BINARY_DIR}/lib/libgc.a" "${CMAKE_BINARY_DIR}/lib/libuv.a" "${CMAKE_BINARY_DIR}/lib/libjansson.a" DESTINATION lib)

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME frontend COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/compiler_cli.py $<TARGET_FILE:lyra> Frontend)
    add_test(NAME native COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/compiler_cli.py $<TARGET_FILE:lyra> Native)
    add_test(NAME objects COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/objects_cli.py $<TARGET_FILE:lyra>)
    add_test(NAME closures COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/closures_cli.py $<TARGET_FILE:lyra>)
    add_test(NAME http COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/http_cli.py $<TARGET_FILE:lyra>)
    foreach(suite IN ITEMS packages modules generics async generics_tasks io sqlite sums macros jsonlines conversions defer exceptions)
        add_test(NAME ${suite} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/${suite}_cli.py $<TARGET_FILE:lyra>)
    endforeach()
    add_test(NAME collections COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/collections_cli.py $<TARGET_FILE:lyra>)
    add_test(NAME context COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/context_cli.py $<TARGET_FILE:lyra>)
    add_test(NAME installed COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/installed_cli.py ${CMAKE_COMMAND} ${CMAKE_BINARY_DIR})
    add_test(NAME sugar COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/sugar_cli.py $<TARGET_FILE:lyra>)
    foreach(component IN ITEMS managed tasks sqlite lists exceptions context task_errors)
        add_executable(test-${component} tests/runtime_${component}.cpp)
        target_link_libraries(test-${component} PRIVATE lyraRuntime)
        add_test(NAME runtime-${component} COMMAND test-${component})
        set_tests_properties(runtime-${component} PROPERTIES TIMEOUT 45)
    endforeach()
endif()
